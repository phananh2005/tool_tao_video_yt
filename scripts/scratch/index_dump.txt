    <script>
        const { createApp } = Vue;

        createApp({
            data() {
                return {
                    currentScreen: 'dashboard',
                    projects: [],
                    newTheme: '',
                    isRabbitHole: false,
                    currentProject: null,
                    currentIdeas: [],
                    scriptData: null,
                    voiceoverData: null,
                    assetData: null,
                    renderPath: null,
                    seoData: null,
                    loading: false,
                    loadingMsg: 'Đang xử lý...'
                }
            },
            mounted() {
                this.loadDashboard();
            },
            methods: {
                async apiCall(url, method = 'GET', body = null) {
                    try {
                        const options = {
                            method,
                            headers: {
                                'Content-Type': 'application/json'
                            }
                        };
                        if (body) options.body = JSON.stringify(body);
                        const res = await fetch("http://127.0.0.1:8000" + url, options);
                        const data = await res.json();
                        if (!res.ok) throw new Error(data.detail || 'API Error');
                        return data;
                    } catch (e) {
                        alert("Lỗi: " + e.message);
                        throw e;
                    }
                },
                withLoading(msg, fn) {
                    this.loading = true;
                    this.loadingMsg = msg;
                    Promise.resolve(fn()).finally(() => {
                        this.loading = false;
                    });
                },
                async loadDashboard() {
                    this.withLoading('Đang tải danh sách dự án...', async () => {
                        this.projects = await this.apiCall('/api/dashboard');
                        this.currentScreen = 'dashboard';
                        this.currentProject = null;
                    });
                },
                goDashboard() {
                    this.loadDashboard();
                },
                async openIdeaBank(p) {
                    this.withLoading('Đang mở Kho Ý Tưởng...', async () => {
                        this.currentProject = { id: p.id, name: p.name };
                        this.currentIdeas = await this.apiCall("/api/projects/" + p.id + "/ideas");
                        this.currentScreen = 'phase1';
                    });
                },
                async deleteMedia(p) {
                    if (!confirm('Bạn có chắc muốn xóa TOÀN BỘ FILE MEDIA (hình ảnh, âm thanh, video) của dự án này để giải phóng dung lượng ổ cứng? Dữ liệu chữ (kịch bản, tiêu đề) vẫn sẽ được giữ lại trong Database để chống trùng lặp.')) return;
                    this.withLoading('Đang xóa file media...', async () => {
                        await this.apiCall("/api/projects/" + p.id + "/media", 'DELETE');
                        alert('Đã xóa sạch file media thành công! Dữ liệu dự án vẫn còn để hệ thống học hỏi.');
                        this.loadDashboard();
                    });
                },
                async rejectIdea(idea) {
                    if (!confirm('Bạn có chắc muốn vứt bỏ ý tưởng này? Nó sẽ không bao giờ xuất hiện lại.')) return;
                    this.withLoading('Đang xóa...', async () => {
                        await this.apiCall("/api/ideas/" + idea.id + "/reject", 'PUT');
                        this.currentIdeas = this.currentIdeas.filter(i => i.id !== idea.id);
                    });
                },
                async brainstorm() {
                    if (!this.newTheme) return;
                    this.withLoading('AI đang động não ý tưởng...', async () => {
                        const data = await this.apiCall('/api/phase1/brainstorm', 'POST', { theme: this.newTheme, is_rabbit_hole: this.isRabbitHole });
                        this.loadIdeas(data.project_id);
                    });
                },
                async loadIdeas(projectId) {
                    this.withLoading('Đang tải danh sách ý tưởng...', async () => {
                        this.currentIdeas = await this.apiCall("/api/projects/" + projectId + "/ideas");
                        this.currentProject = { id: projectId, name: this.newTheme || 'Dự án đang mở' };
                        this.currentScreen = 'phase1';
                    });
                },
                async resumeProject(p) {
                    this.currentProject = { id: p.id, name: p.name, current_script_id: p.current_script_id };
                    if (p.phase === 1 || p.phase === '1 (Ý tưởng)') {
                        await this.loadIdeas(p.id);
                    } else if (p.phase === 2 || p.phase === '2 (Kịch bản)') {
                        this.loadScript(p.current_script_id);
                    } else if (p.phase === 3 || p.phase === '3 (Voiceover)') {
                        this.loadVoiceover(p.current_script_id);
                    } else if (p.phase === 4 || p.phase === '4 (Hình ảnh)') {
                        this.loadAssets(p.current_script_id);
                    } else if (p.phase === 5 || p.phase === '5 (Render)') {
                        this.currentScreen = 'phase5';
                    } else if (p.phase === 6 || p.phase === '6 (SEO & Publish)') {
                        this.loadSEO(p.current_script_id);
                    } else {
                        await this.loadIdeas(p.id);
                    }
                },
                async generateScript(ideaId) {
                    this.withLoading('Đang tạo kịch bản...', async () => {
                        const data = await this.apiCall("/api/phase2/generate/" + ideaId, 'POST');
                        this.currentProject.current_script_id = data.script_id;
                        this.scriptData = data.script;
                        this.currentScreen = 'phase2';
                    });
                },
                async loadScript(scriptId) {
                    this.withLoading('Đang tải kịch bản...', async () => {
                        this.scriptData = await this.apiCall("/api/scripts/" + scriptId);
                        this.currentScreen = 'phase2';
                    });
                },
                async saveScript() {
                    this.withLoading('Đang lưu kịch bản...', async () => {
                        await this.apiCall("/api/scripts/" + this.currentProject.current_script_id, 'PUT', { content: this.scriptData });
                        alert('Lưu kịch bản thành công!');
                    });
                },
                async generateVoiceover() {
                    this.withLoading('Đang tạo lời thoại...', async () => {
                        const data = await this.apiCall("/api/phase3/generate/" + this.currentProject.current_script_id, 'POST');
                        this.voiceoverData = data.voiceover;
                        this.currentScreen = 'phase3';
                    });
                },
                async loadVoiceover(scriptId) {
                    this.withLoading('Đang tải lời thoại...', async () => {
                        this.voiceoverData = await this.apiCall("/api/voiceovers/" + scriptId);
                        this.currentScreen = 'phase3';
                    });
                },
                async saveVoiceover() {
                    this.withLoading('Đang lưu lời thoại...', async () => {
                        await this.apiCall("/api/voiceovers/" + this.currentProject.current_script_id, 'PUT', { content: this.voiceoverData });
                        alert('Lưu lời thoại thành công!');
                    });
                },
                async synthesizeVoice() {
                    this.withLoading('Đang tổng hợp âm thanh...', async () => {
                        await this.apiCall("/api/phase3_5/synthesize/" + this.currentProject.current_script_id, 'POST');
                        alert('Đã thu âm/tổng hợp (Text Mocked)!');
                    });
                },
                async generateAssets() {
                    this.withLoading('Đang vẽ hình ảnh...', async () => {
                        await this.apiCall("/api/phase4/generate/" + this.currentProject.current_script_id, 'POST');
                        await this.loadAssets(this.currentProject.current_script_id);
                    });
                },
                async loadAssets(scriptId) {
                    this.withLoading('Đang tải hình ảnh...', async () => {
                        this.assetData = await this.apiCall("/api/assets/" + scriptId);
                        this.currentScreen = 'phase4';
                    });
                },
                async saveAssets() {
                    this.withLoading('Đang lưu hình ảnh...', async () => {
                        await this.apiCall("/api/assets/" + this.currentProject.current_script_id, 'PUT', { content: this.assetData });
                        alert('Lưu hình ảnh thành công!');
                    });
                },
                async renderVideo() {
                    this.withLoading('Đang render video bằng FFmpeg...', async () => {
                        const res = await this.apiCall("/api/phase5/render/" + this.currentProject.current_script_id, 'POST');
                        this.renderPath = res.video_path;
                        this.currentScreen = 'phase5';
                    });
                },
                async generateSEO() {
                    this.withLoading('Đang tạo Metadata SEO...', async () => {
                        const res = await this.apiCall("/api/phase6/generate/" + this.currentProject.current_script_id, 'POST');
                        this.seoData = res.seo;
                        this.currentScreen = 'phase6';
                    });
                },
                async loadSEO(scriptId) {
                    this.withLoading('Đang tải Metadata SEO...', async () => {
                        this.seoData = await this.apiCall("/api/seo/" + scriptId);
                        this.currentScreen = 'phase6';
                    });
                },
                copySEO() {
                    const text = "TITLE:\n" + this.seoData.title + "\n\nDESC:\n" + this.seoData.description + "\n\nTAGS:\n" + this.seoData.tags.join(', ');
                    navigator.clipboard.writeText(text);
                    alert('Đã copy!');
                }
            }
        }).mount('#app');
